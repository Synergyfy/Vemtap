        function selectGoal(element) {
            // Remove active class from all cards
            const cards = document.querySelectorAll('.selection-card');
            cards.forEach(card => {
                card.classList.remove('active', 'border-2', 'border-primary');
                card.classList.add('border', 'border-outline-variant');

                iconContainer.classList.remove('bg-primary', 'text-white');
                iconContainer.classList.add('bg-surface-container');

                const check = card.querySelector('.mt-auto');
                if (check) check.remove();
            });

            // Add active class to clicked card

            element.classList.remove('border', 'border-outline-variant');

            // Add checkmark indicator

            if (window.navigator.vibrate) {
                window.navigator.vibrate(10);
            }
        }

        function toggleBottomSheet(show) {
            const sheet = document.getElementById('bottom-sheet');
            const backdrop = document.getElementById('backdrop');
            
            if (show) {
                backdrop.classList.remove('hidden');
                setTimeout(() => {
                    backdrop.classList.add('opacity-100');
                    sheet.classList.remove('translate-y-full');
                    sheet.classList.add('translate-y-0');
                }, 10);
            } else {
                sheet.classList.remove('translate-y-0');
                sheet.classList.add('translate-y-full');
                backdrop.classList.remove('opacity-100');
                setTimeout(() => {
                    backdrop.classList.add('hidden');
                }, 400);
            }
        }

        function applyAndClose() {
            // Mock content application
            const headlineInput = document.querySelector('input[placeholder*="Summer"]');
            if(headlineInput) headlineInput.value = "Unleash Your Summer: The Definitive Collection 2024";
            
            toggleBottomSheet(false);
            
            // Visual feedback
            const btn = document.querySelector('button[onclick*="toggleBottomSheet"]');
            btn.innerHTML = '<span class="material-symbols-outlined">check_circle</span> Optimized!';
            btn.classList.replace('text-primary', 'text-secondary');
        }

        // Ripple Effect Helper
        document.querySelectorAll('button').forEach(button => {
            button.addEventListener('mousedown', function(e) {
                const rect = this.getBoundingClientRect();
                const x = e.clientX - rect.left;
                const y = e.clientY - rect.top;
                
                const ripple = document.createElement('span');
                ripple.style.left = `${x}px`;
                ripple.style.top = `${y}px`;
                ripple.className = 'absolute bg-white/30 rounded-full w-2 h-2 animate-ping pointer-events-none';
                
                this.style.position = 'relative';
                this.style.overflow = 'hidden';
                this.appendChild(ripple);
                
                setTimeout(() => ripple.remove(), 1000);
            });

        function selectBranch(element) {
            // Remove selection from all cards
            document.querySelectorAll('.branch-card').forEach(card => {
                card.classList.remove('selected', 'border-primary', 'bg-primary-container/5', 'shadow-lg');
                card.classList.add('border-transparent');
            });

            // Add selection to clicked card
            element.classList.add('selected', 'border-primary', 'bg-primary-container/5', 'shadow-lg');
            element.classList.remove('border-transparent');

            // Update details section
            const branchName = element.getAttribute('data-branch');
            const detailsSection = document.getElementById('branch-details');
            const activeNameLabel = document.getElementById('active-branch-name');
            const continueBtn = document.getElementById('continue-btn');

            activeNameLabel.innerText = branchName;
            detailsSection.classList.remove('opacity-0', 'translate-y-4');
            detailsSection.classList.add('opacity-100', 'translate-y-0');

            // Enable continue button
            continueBtn.disabled = false;
            continueBtn.classList.remove('bg-outline-variant', 'opacity-50', 'cursor-not-allowed');
            continueBtn.classList.add('bg-primary', 'opacity-100', 'cursor-pointer');

            // Simple micro-interaction
            element.style.transform = 'scale(0.98)';
            setTimeout(() => {
                element.style.transform = 'scale(1)';
            }, 100);
        }

                    alert('Branch selected successfully! Redirecting to dashboard...');

            }
        });

        const steps = [
            { id: 1, duration: 1500, label: 'Initializing...' },
            { id: 2, duration: 2000, label: 'Encoding...' },
            { id: 3, duration: 2500, label: 'Styling...' },
            { id: 4, duration: 1500, label: 'Finalizing...' }
        ];

        let currentProgress = 0;
        const progressBar = document.getElementById('progress-bar');
        const progressPercent = document.getElementById('progress-percent');
        const statusText = document.getElementById('status-text');

        async function runProgress() {
            for (const step of steps) {
                // Activate current step
                const row = document.getElementById(`step-${step.id}`);
                const loader = document.getElementById(`loader-${step.id}`);
                const check = document.getElementById(`check-1`); // Placeholder ref

                row.classList.remove('opacity-40');
                row.classList.add('bg-primary/5', 'border-primary/20');
                loader.classList.remove('hidden');
                statusText.innerText = step.label;

                // Animate progress bar incrementally
                const increment = 25;
                const target = currentProgress + increment;
                
                // Duration for the sub-step
                const duration = step.duration;
                const interval = 20;
                const stepsCount = duration / interval;
                const progressStep = increment / stepsCount;

                for (let i = 0; i < stepsCount; i++) {
                    currentProgress += progressStep;
                    progressBar.style.width = `${Math.min(currentProgress, target)}%`;
                    progressPercent.innerText = `${Math.round(currentProgress)}%`;
                    await new Promise(resolve => setTimeout(resolve, interval));
                }

                // Complete current step
                loader.classList.add('hidden');
                const checkIcon = document.getElementById(`check-${step.id}`);
                checkIcon.classList.remove('hidden');
                checkIcon.classList.add('text-primary');
                row.classList.remove('bg-primary/5');
                row.classList.add('bg-surface-container-lowest');
            }
            
            statusText.innerText = 'Ready!';
            statusText.classList.add('animate-bounce');
        }

        document.addEventListener('DOMContentLoaded', runProgress);
    </script>

            const container = document.getElementById('main-preview-container');

                container.classList.remove('scale-100');
                container.classList.add('scale-110');
            }
